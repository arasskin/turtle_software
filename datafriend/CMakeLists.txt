cmake_minimum_required(VERSION 3.18)
project(datafriend LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

# Keep vendor compilation separate from the store and its language bindings.
add_library(datafriend_storage OBJECT
  ../third_party/aelmdb/src/mdb.c ../third_party/aelmdb/src/midl.c)
target_compile_definitions(datafriend_storage PRIVATE MDB_HASH_SIZE=8)
set_target_properties(datafriend_storage PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(datafriend_crypto OBJECT
  ../third_party/monocypher/monocypher.c ../third_party/monocypher/monocypher-ed25519.c)
set_target_properties(datafriend_crypto PROPERTIES POSITION_INDEPENDENT_CODE ON)

option(DATAFRIEND_SHARED "Build a shared library" ON)
if(DATAFRIEND_SHARED)
  set(DATAFRIEND_LIBRARY_TYPE SHARED)
else()
  set(DATAFRIEND_LIBRARY_TYPE STATIC)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  target_compile_definitions(datafriend_storage PRIVATE MDB_USE_POSIX_SEM=1)
endif()

add_library(datafriend ${DATAFRIEND_LIBRARY_TYPE} db.c memory.c flush.c engine.c query.c result.c sync.c identity.c network.c)
add_library(datafriend::datafriend ALIAS datafriend)
target_include_directories(datafriend
  PUBLIC ..
  PRIVATE . ../littlebear)
target_include_directories(datafriend SYSTEM PRIVATE
  ../third_party/aelmdb/src ../third_party/monocypher ../third_party/xxhash)
target_compile_definitions(datafriend PRIVATE MDB_HASH_SIZE=8)
target_link_libraries(datafriend PRIVATE datafriend_storage datafriend_crypto Threads::Threads)
target_compile_options(datafriend PRIVATE -Wall -Wextra -pedantic-errors)
if(APPLE)
  target_sources(datafriend PRIVATE keychain.c)
  target_link_libraries(datafriend PRIVATE "-framework Security" "-framework CoreFoundation")
elseif(ANDROID)
  target_sources(datafriend PRIVATE keystore.c)
endif()

option(DATAFRIEND_TESTS "Build Datafriend correctness tests" ON)
if(DATAFRIEND_TESTS AND NOT ANDROID AND NOT IOS)
  enable_language(CXX)
  enable_testing()
  target_compile_definitions(datafriend PUBLIC DATAFRIEND_TESTING=1)
  add_executable(transport_test tests/transport_test.cpp)
  target_link_libraries(transport_test PRIVATE datafriend::datafriend)
  add_executable(db_test tests/db_test.c)
  target_link_libraries(db_test PRIVATE datafriend::datafriend)
  target_compile_options(db_test PRIVATE -Wall -Wextra -pedantic-errors)
  add_test(NAME db_test COMMAND db_test)
  add_executable(c99_test tests/c99_test.c)
  target_link_libraries(c99_test PRIVATE datafriend::datafriend)
  target_include_directories(c99_test PRIVATE . ../third_party/xxhash)
  target_compile_options(c99_test PRIVATE -Wall -Wextra -pedantic-errors)
  add_test(NAME c99_test COMMAND c99_test)
  add_executable(network_queue_test tests/network_queue_test.c)
  target_link_libraries(network_queue_test PRIVATE datafriend::datafriend)
  target_include_directories(network_queue_test PRIVATE .)
  target_include_directories(network_queue_test SYSTEM PRIVATE ../third_party/aelmdb/src ../third_party/xxhash)
  target_compile_definitions(network_queue_test PRIVATE MDB_HASH_SIZE=8)
  target_compile_options(network_queue_test PRIVATE -Wall -Wextra -pedantic-errors)
  add_test(NAME network_queue_test COMMAND network_queue_test)
  add_executable(channel_test tests/channel_test.c)
  target_link_libraries(channel_test PRIVATE datafriend::datafriend)
  target_include_directories(channel_test PRIVATE . ../third_party/xxhash)
  target_compile_options(channel_test PRIVATE -Wall -Wextra -pedantic-errors)
  foreach(name engine_test async_flush_test network_test identity_test slab_test)
    add_executable(${name} tests/${name}.cpp)
    target_link_libraries(${name} PRIVATE datafriend::datafriend)
    target_include_directories(${name} PRIVATE . ../third_party/aelmdb/src ../littlebear ../third_party/monocypher)
    target_compile_definitions(${name} PRIVATE MDB_HASH_SIZE=8)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()
