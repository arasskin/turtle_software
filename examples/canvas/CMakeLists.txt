cmake_minimum_required(VERSION 3.18)
project(turtle_canvas LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(DATAFRIEND_TESTS OFF CACHE BOOL "" FORCE)
set(DATAFRIEND_SHARED OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../../datafriend datafriend)
add_executable(janet ../../third_party/janet/janet.c ../../third_party/janet/shell.c)
target_include_directories(janet PRIVATE ../../third_party/janet)
target_compile_definitions(janet PRIVATE JANET_SIMPLE_GETLINE)
set_target_properties(janet PROPERTIES ENABLE_EXPORTS ON)
target_link_libraries(janet PRIVATE m pthread ${CMAKE_DL_LIBS})
add_library(canvas_store MODULE store.c)
target_include_directories(canvas_store PRIVATE ../../third_party/janet)
target_link_libraries(canvas_store PRIVATE datafriend::datafriend)
target_compile_options(canvas_store PRIVATE -Wall -Wextra -Werror)
set_target_properties(canvas_store PROPERTIES PREFIX "" SUFFIX ".so")
if(APPLE)
  target_link_options(canvas_store PRIVATE "-undefined" "dynamic_lookup")
endif()
